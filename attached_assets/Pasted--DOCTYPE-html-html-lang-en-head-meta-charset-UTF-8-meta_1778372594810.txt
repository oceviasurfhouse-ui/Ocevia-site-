<script>

/* SUPABASE */
const supabase = supabase.createClient(
"https://YOUR_PROJECT.supabase.co",
"YOUR_KEY"
);

/* STRIPE */
const stripe = Stripe("YOUR_STRIPE_PUBLIC_KEY");

/* BOOKING */
async function book(){

let name = document.getElementById("name").value;
let email = document.getElementById("email").value;
let room = document.getElementById("room").value;
let days = document.getElementById("days").value;

if(!name||!email||!days) return alert("Fill all");

let price = days * 90;

const {error} = await supabase.from("bookings").insert([
{name,email,room,days,price}
]);

if(!error){
document.getElementById("result").innerText =
"Booked ✔ Price: "+price+"€";

sendEmail(name,email,room,days,price);
loadDashboard();
}
}

/* EMAIL API */
function sendEmail(name,email,room,days,price){
fetch("/api/email",{
method:"POST",
headers:{"Content-Type":"application/json"},
body:JSON.stringify({name,email,room,days,price})
});
}

/* DASHBOARD */
async function loadDashboard(){
const {data} = await supabase.from("bookings").select("*");
if(data){
let total=0;
data.forEach(b=>total+=b.price);
document.getElementById("count").innerText=data.length;
document.getElementById("revenue").innerText=total+"€";
}
}
loadDashboard();

/* STRIPE */
function payNow(){
fetch("/api/checkout")
.then(r=>r.json())
.then(d=>window.location.href=d.url);
}

/* CHAT AI */
async function sendMsg(){
let msg=document.getElementById("msg").value;
let box=document.getElementById("chatBody");

box.innerHTML+=`<p>You: ${msg}</p>`;

const res = await fetch("/api/chat",{
method:"POST",
headers:{"Content-Type":"application/json"},
body:JSON.stringify({message:msg})
});

const data = await res.json();

box.innerHTML+=`<p>AI: ${data.reply}</p>`;
}

function toggleChat(){
let c=document.getElementById("chat");
c.style.display=c.style.display==="flex"?"none":"flex";
c.style.flexDirection="column";
}

function scrollToBook(){
document.getElementById("booking").scrollIntoView({behavior:"smooth"});
}

</script>
